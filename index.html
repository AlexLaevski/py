'''1:
сорок один и тридцать одна сотая разделить на семнадцать → два и сорок три сотых

2:
девятнадцать и восемьдесят две сотых разделить на девяносто девять → ноль и двадцать сотых и ноль два в периоде

3:
пять плюс два умножить на три минус один → десять

4:
скобка открывается пять плюс два скобка закрывается умножить на три минус один → двадцать'''



from fractions import Fraction

# ---------------------------------------------------------------- словари

UNITS = {
    "ноль": 0, "один": 1, "одна": 1, "одно": 1, "два": 2, "две": 2,
    "три": 3, "четыре": 4, "пять": 5, "шесть": 6, "семь": 7,
    "восемь": 8, "девять": 9,
}
TEENS = {
    "десять": 10, "одиннадцать": 11, "двенадцать": 12, "тринадцать": 13,
    "четырнадцать": 14, "пятнадцать": 15, "шестнадцать": 16,
    "семнадцать": 17, "восемнадцать": 18, "девятнадцать": 19,
}
TENS = {
    "двадцать": 20, "тридцать": 30, "сорок": 40, "пятьдесят": 50,
    "шестьдесят": 60, "семьдесят": 70, "восемьдесят": 80, "девяносто": 90,
}
HUNDREDS = {
    "сто": 100, "двести": 200, "триста": 300, "четыреста": 400,
    "пятьсот": 500, "шестьсот": 600, "семьсот": 700, "восемьсот": 800,
    "девятьсот": 900,
}
SCALES = {
    "тысяча": 10**3, "тысячи": 10**3, "тысяч": 10**3,
    "миллион": 10**6, "миллиона": 10**6, "миллионов": 10**6,
}
NUMBER_WORDS = set(UNITS) | set(TEENS) | set(TENS) | set(HUNDREDS) | set(SCALES)

# знаков после запятой -> (форма для "одна ...", форма для остальных)
FRACTION_NAMES = {
    1: ("десятая", "десятых"),
    2: ("сотая", "сотых"),
    3: ("тысячная", "тысячных"),
    4: ("десятитысячная", "десятитысячных"),
    5: ("стотысячная", "стотысячных"),
    6: ("миллионная", "миллионных"),
}
FRACTION_DIGITS = {form: d for d, forms in FRACTION_NAMES.items() for form in forms}

MAX_DIGITS = 6   # дробная часть до миллионных включительно
MAX_PERIOD = 4   # период до четырех знаков

# фразы операций; длинные варианты идут раньше коротких
OPERATOR_PHRASES = [
    (("остаток", "от", "деления", "на"), "%"),
    (("остаток", "от", "деления"), "%"),
    (("скобка", "открывается"), "("),
    (("скобка", "закрывается"), ")"),
    (("умножить", "на"), "*"),
    (("умножить",), "*"),
    (("разделить", "на"), "/"),
    (("разделить",), "/"),
    (("плюс",), "+"),
    (("минус",), "-"),
]

PRECEDENCE = {"+": 1, "-": 1, "*": 2, "/": 2, "%": 2, "neg": 3}


# ------------------------------------------------------- слова -> число

def _parse_triple(words):
    """Число от 1 до 999: [сотни] [десятки] [единицы] или [сотни] [10..19]."""
    value, last_rank, after_teen = 0, 4, False
    for w in words:
        if after_teen:
            raise ValueError("неверная запись числа: «%s»" % " ".join(words))
        if w in HUNDREDS:
            rank, v = 3, HUNDREDS[w]
        elif w in TENS:
            rank, v = 2, TENS[w]
        elif w in TEENS:
            rank, v, after_teen = 1.5, TEENS[w], True
        elif w in UNITS and w != "ноль":
            rank, v = 1, UNITS[w]
        else:
            raise ValueError("неверная запись числа: «%s»" % " ".join(words))
        if rank >= last_rank or (rank == 1.5 and last_rank == 2):
            raise ValueError("неверная запись числа: «%s»" % " ".join(words))
        value += v
        last_rank = rank
    return value


def parse_int(words):
    """Целое число от 0 до 999 999 999, записанное словами."""
    if words == ["ноль"]:
        return 0
    total, chunk, prev_scale = 0, [], None
    for w in words:
        if w in SCALES:
            scale = SCALES[w]
            if prev_scale is not None and scale >= prev_scale:
                raise ValueError("неверная запись числа: «%s»" % " ".join(words))
            total += (_parse_triple(chunk) if chunk else 1) * scale
            chunk, prev_scale = [], scale
        else:
            chunk.append(w)
    if chunk:
        total += _parse_triple(chunk)
    return total


# ------------------------------------------------------- число -> слова

_UNITS_M = ["", "один", "два", "три", "четыре", "пять", "шесть", "семь", "восемь", "девять"]
_UNITS_F = ["", "одна", "две"] + _UNITS_M[3:]
_TEENS = ["десять", "одиннадцать", "двенадцать", "тринадцать", "четырнадцать",
          "пятнадцать", "шестнадцать", "семнадцать", "восемнадцать", "девятнадцать"]
_TENS = ["", "", "двадцать", "тридцать", "сорок", "пятьдесят", "шестьдесят",
         "семьдесят", "восемьдесят", "девяносто"]
_HUNDREDS = ["", "сто", "двести", "триста", "четыреста", "пятьсот", "шестьсот",
             "семьсот", "восемьсот", "девятьсот"]
_SCALE_NAMES = [
    (10**12, ("триллион", "триллиона", "триллионов"), False),
    (10**9, ("миллиард", "миллиарда", "миллиардов"), False),
    (10**6, ("миллион", "миллиона", "миллионов"), False),
    (10**3, ("тысяча", "тысячи", "тысяч"), True),
]


def _plural(n, forms):
    """forms = (1 тысяча, 2 тысячи, 5 тысяч)."""
    if 11 <= n % 100 <= 19:
        return forms[2]
    if n % 10 == 1:
        return forms[0]
    if 2 <= n % 10 <= 4:
        return forms[1]
    return forms[2]


def _triple_words(n, feminine):
    words = []
    h, rest = divmod(n, 100)
    if h:
        words.append(_HUNDREDS[h])
    if 10 <= rest <= 19:
        words.append(_TEENS[rest - 10])
    else:
        t, u = divmod(rest, 10)
        if t:
            words.append(_TENS[t])
        if u:
            words.append((_UNITS_F if feminine else _UNITS_M)[u])
    return words


def int_to_words(n, feminine=False):
    """Неотрицательное целое -> слова ("одна"/"две" при feminine=True)."""
    if n == 0:
        return "ноль"
    if n >= 10**15:
        raise ValueError("слишком большое число для записи словами")
    words = []
    for size, forms, fem in _SCALE_NAMES:
        chunk, n = divmod(n, size)
        if chunk:
            words += _triple_words(chunk, fem)
            words.append(_plural(chunk, forms))
    if n:
        words += _triple_words(n, feminine)
    return " ".join(words)


def _fraction_words(numerator, digits):
    """numerator / 10**digits -> "сорок три сотых", "одна десятая"."""
    singular, plural = FRACTION_NAMES[digits]
    one = numerator % 10 == 1 and numerator % 100 != 11
    return "%s %s" % (int_to_words(numerator, feminine=True), singular if one else plural)


def _period_words(digits):
    """'02' -> "ноль два": ведущие нули читаем отдельно, остальное числом."""
    rest = digits.lstrip("0")
    words = ["ноль"] * (len(digits) - len(rest))
    if rest:
        words.append(int_to_words(int(rest)))
    return " ".join(words)


def _strip_zeros(numerator, digits):
    while digits and numerator % 10 == 0:
        numerator //= 10
        digits -= 1
    return numerator, digits


def _power_split(den):
    """den = 2**a * 5**b * m -> (max(a, b), m)."""
    a = b = 0
    while den % 2 == 0:
        den //= 2
        a += 1
    while den % 5 == 0:
        den //= 5
        b += 1
    return max(a, b), den


def format_number(x):
    """Fraction -> текст ответа."""
    negative = x < 0
    x = abs(x)
    whole, frac = divmod(x, 1)
    whole = int(whole)
    pre_len, rest = _power_split(frac.denominator)

    if frac == 0:
        parts = []
    elif rest == 1 and pre_len <= MAX_DIGITS:
        # конечная дробь, помещается в миллионные
        parts = [_fraction_words(int(frac * 10**pre_len), pre_len)]
    else:
        period_len = next((L for L in range(1, MAX_PERIOD + 1) if pow(10, L, rest) == 1), None)
        if rest > 1 and period_len and pre_len <= MAX_DIGITS:
            # периодическая дробь: 0,20(02)
            pre = int(frac * 10**pre_len)
            period = int(frac * 10**(pre_len + period_len)) - pre * 10**period_len
            parts = []
            if pre_len:
                parts.append(_fraction_words(pre, pre_len))
            parts.append(_period_words(str(period).zfill(period_len)) + " в периоде")
        else:
            # округляем до миллионных (половину — вверх)
            scaled = int(x * 10**MAX_DIGITS + Fraction(1, 2))
            whole, numerator = divmod(scaled, 10**MAX_DIGITS)
            numerator, digits = _strip_zeros(numerator, MAX_DIGITS)
            parts = [_fraction_words(numerator, digits)] if numerator else []

    text = " и ".join([int_to_words(whole)] + parts)
    if negative and (whole or parts):
        text = "минус " + text
    return text


# ------------------------------------------------------- разбор выражения

def tokenize(text):
    """Строка -> список токенов: Fraction (число) или строка-операция."""
    words = text.lower().replace("ё", "е").split()
    tokens, i = [], 0
    while i < len(words):
        for phrase, op in OPERATOR_PHRASES:
            if tuple(words[i:i + len(phrase)]) == phrase:
                tokens.append(op)
                i += len(phrase)
                break
        else:
            if words[i] not in NUMBER_WORDS:
                raise ValueError("непонятное слово: «%s»" % words[i])
            j = i
            while j < len(words) and words[j] in NUMBER_WORDS:
                j += 1
            start, i = i, j
            value = Fraction(parse_int(words[start:i]))
            # дробь без целой части: "три десятых" = "ноль и три десятых"
            if i < len(words) and words[i] in FRACTION_DIGITS:
                digits = FRACTION_DIGITS[words[i]]
                if value >= 10**digits:
                    raise ValueError("дробь больше единицы: «%s»" % " ".join(words[start:i + 1]))
                value /= 10**digits
                i += 1
            # дробная часть: "... и тридцать одна сотая"
            elif i < len(words) and words[i] == "и":
                j = i + 1
                while j < len(words) and words[j] in NUMBER_WORDS:
                    j += 1
                if j == i + 1 or j >= len(words) or words[j] not in FRACTION_DIGITS:
                    raise ValueError("после «и» ожидается дробная часть, например «сорок три сотых»")
                numerator = parse_int(words[i + 1:j])
                digits = FRACTION_DIGITS[words[j]]
                if numerator >= 10**digits:
                    raise ValueError("дробная часть больше единицы: «%s»" % " ".join(words[i + 1:j + 1]))
                value += Fraction(numerator, 10**digits)
                i = j + 1
            tokens.append(value)
    return tokens


def to_rpn(tokens):
    """Алгоритм сортировочной станции: инфиксная запись -> обратная польская."""
    output, stack = [], []
    expect_operand = True
    for tok in tokens:
        if isinstance(tok, Fraction):
            if not expect_operand:
                raise ValueError("два числа подряд без операции")
            output.append(tok)
            expect_operand = False
        elif tok == "(":
            if not expect_operand:
                raise ValueError("перед скобкой нужна операция")
            stack.append(tok)
        elif tok == ")":
            if expect_operand:
                raise ValueError("пустые скобки или операция перед закрывающей скобкой")
            while stack and stack[-1] != "(":
                output.append(stack.pop())
            if not stack:
                raise ValueError("лишняя закрывающая скобка")
            stack.pop()
        elif expect_operand:
            if tok != "-":
                raise ValueError("операция без левого числа")
            stack.append("neg")  # унарный минус: "минус пять плюс два"
        else:
            while stack and stack[-1] != "(" and PRECEDENCE[stack[-1]] >= PRECEDENCE[tok]:
                output.append(stack.pop())
            stack.append(tok)
            expect_operand = True
    if expect_operand:
        raise ValueError("выражение не закончено")
    while stack:
        op = stack.pop()
        if op == "(":
            raise ValueError("не закрыта скобка")
        output.append(op)
    return output


def eval_rpn(rpn):
    stack = []
    for tok in rpn:
        if isinstance(tok, Fraction):
            stack.append(tok)
        elif tok == "neg":
            stack.append(-stack.pop())
        else:
            b, a = stack.pop(), stack.pop()
            if tok in "/%" and b == 0:
                raise ValueError("деление на ноль")
            if tok == "+":
                stack.append(a + b)
            elif tok == "-":
                stack.append(a - b)
            elif tok == "*":
                stack.append(a * b)
            elif tok == "/":
                stack.append(a / b)
            else:
                stack.append(a % b)
    return stack[0]


def calc(expression):
    """calc("пять плюс два умножить на три минус один") -> "десять"."""
    tokens = tokenize(expression)
    if not tokens:
        raise ValueError("пустое выражение")
    return format_number(eval_rpn(to_rpn(tokens)))


if __name__ == "__main__":
    import sys

    if len(sys.argv) > 1:
